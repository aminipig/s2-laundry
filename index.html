document.getElementById('finance-form').addEventListener('submit', async function(e) { e.preventDefault(); const editId = document.getElementById('edit-id').value; const itemType = document.getElementById('type').value; const inputDateValue = document.getElementById('date').value; const now = new Date(); const hours = String(now.getHours()).padStart(2, '0'); const minutes = String(now.getMinutes()).padStart(2, '0'); const seconds = String(now.getSeconds()).padStart(2, '0'); const fullDateTimeStampString = `${inputDateValue} ${hours}:${minutes}:${seconds}`; const payload = { date: fullDateTimeStampString, type: itemType, description: document.getElementById('description').value, amount: parseFloat(document.getElementById('amount').value), notes: document.getElementById('notes').value, paymentMethod: document.getElementById('payment-method').value }; // แปลงประเภทเป็นข้อความภาษาไทยสำหรับใส่ใน Log const typeLabel = itemType === 'income' ? 'รายรับ' : 'รายจ่าย'; try { if (editId) { const existingItem = window.transactions.find(t => t.id === editId); if (existingItem && existingItem.date.substring(0,10) === inputDateValue) { payload.date = existingItem.date; } const { error } = await _supabase.from('s2_records').update(payload).eq('id', editId); if (error) throw error; // 📝 Log ชัดเจนกรณีแก้ไข await writeActivityLog( `แก้ไขรายการ${typeLabel}`, `แก้ไขรายการเป็น [${typeLabel}]: "${payload.description}" ยอด ${payload.amount.toFixed(2)} บ. (${payload.paymentMethod})` ); } else { const { error } = await _supabase.from('s2_records').insert([payload]); if (error) throw error; // 📝 Log ชัดเจนกรณีเพิ่มใหม่ (ระบุว่าเป็น รายรับ หรือ รายจ่าย) await writeActivityLog( `เพิ่มรายการ${typeLabel}`, `เพิ่มรายการใหม่ [${typeLabel}]: "${payload.description}" ยอด ${payload.amount.toFixed(2)} บ. (${payload.paymentMethod})` ); } window.resetFormMode(); } catch (error) { alert("บันทึกข้อมูลล้มเหลว: " + error.message); } });
